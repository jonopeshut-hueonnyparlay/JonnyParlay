# ── Hard dependencies ───────────────────────────────────────────────────────
# Cross-process file locking. Used for pick_log write safety (run_picks +

# HTTP client for The Odds API + Discord webhooks.
requests

# backports.zoneinfo; python_version < "3.9"
# Required by stdlib zoneinfo on Linux/Docker (Windows uses OS TZ data).
tzdata

# ── Optional dependencies ───────────────────────────────────────────────────

# HTML parsing fallback for the umpire run-environment scrape in
# engine/context_research_v2.py. Soft dependency — the umpire factor degrades
